# Team state can change only when a lineup has already been observed, not from this map's unseen starters.
def bayes(hazard,sigma,reset):
 grid=np.linspace(-2,2,81);prior=np.exp(-.5*(grid/sigma)**2);prior/=prior.sum();state={};known={};used={};out=np.zeros(len(g))
 for day,ix in days:
  active=set(g.loc[ix,'a'])|set(g.loc[ix,'b'])
  for a in active:
   st=state.get(a,prior.copy());st=(1-hazard)*st+hazard*prior
   if reset and used.get(a)!=known.get(a):st=prior.copy()
   state[a]=st;used[a]=known.get(a)
  mu={a:float(state[a]@grid) for a in active};ll=defaultdict(lambda:np.zeros(len(grid)))
  for i in ix:
   a,b=ids[i];pa=expit(eta[i]+grid-mu[b]);pb=expit(eta[i]+mu[a]-grid)
   # Integrate independent posterior team uncertainties for the forecast.
   out[i]=float(state[a]@expit(eta[i]+grid[:,None]-grid[None,:])@state[b])
   ll[a]+=np.log(np.clip(pa if y[i] else 1-pa,1e-15,1));ll[b]+=np.log(np.clip(pb if y[i] else 1-pb,1e-15,1))
  for a,v in ll.items():
   w=state[a]*np.exp(v-v.max());state[a]=w/w.sum()
  for r in obsdays.get(day,[]):known[r.teamid]=rosters.get((day,r.gameid,r.teamid),())
 return out
# Observation model is fit only on earlier calendar years, excluding held-out year's outcomes.
gd=obs[obs.teamid.isin(g.a)].set_index(['gameid','teamid']).gd15;gda=np.array([gd.get((r.gameid,r.a),np.nan) for r in g.itertuples()]);measure={}
for year in sorted(g.year.unique()):

 if tr.sum()>100:
  beta=np.dot(eta[tr],gda[tr])/np.dot(eta[tr],eta[tr]);sigma=np.std(gda[tr]-beta*eta[tr]);measure[int(year)]=(float(beta),float(sigma))
 else:measure[int(year)]=(1000.,3000.)
def score(alpha,gamma):
 state=defaultdict(float);out=np.zeros(len(g))
 for day,ix in days:
  updates=defaultdict(float)
  for i in ix:
   a,b=ids[i];e=eta[i]+state[a]-state[b];p=expit(e);out[i]=p;u=alpha*(y[i]-p)
   if gamma and np.isfinite(gda[i]):
    be,si=measure[int(g.year.iloc[i])];u+=gamma*np.clip((gda[i]-be*e)/si,-3,3)
   updates[a]+=u;updates[b]-=u
  for a,v in updates.items():state[a]=float(np.clip(.995*state[a]+v,-1.5,1.5))
 return out
