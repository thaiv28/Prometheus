def norm(s):
 s=re.sub('[^a-z0-9]','',str(s).lower());return {'monkeyking':'wukong','renataglasc':'renata','nunuandwillump':'nunu'}.get(s,s)
def patchnorm(x):

(O/'patch-source-audit.json').write_text(json.dumps({'rows':len(wr),'zero_winrate_and_pickrate':int(zero.sum()),'by_year':{str(k):int(v) for k,v in zero.groupby(wr.date.dt.year).sum().items()}},indent=2))

def rate(pa,ch,role,day,lag):
 a=wrmap.get((pa,ch,role),());v=None
 for d,w in a:
  if d<=day-lag:v=w

# Style EWMA: tempo and objective rate; zero until observed, with missingness controls.
roleorder=['top','jng','mid','bot','sup']
def teamfeat_full(team,pa,day,actual=None):
 day=int(day.value//86400000000000)
 roster=rosters.get(team,{});out={};ent=[];ex=[]
 for role,pl in roster.items():
  h=hist[pl];counts=Counter(ch for ch,rr,gd,home in h if rr==role)
  n=sum(counts.values());probs=np.array(list(counts.values()),float)/max(n,1);ent.append(float(-sum(probs*np.log(probs+1e-15))));ex.append(np.log1p(n))
 out['pool_entropy']=np.mean(ent) if ent else 0;out['experience']=np.mean(ex) if ex else 0
 # Higher-order overlap / familiarity controls use last observed roster only.
 line=tuple(sorted(roster.values()));out['line_games']=np.log1p(seenline[line]);out['roster_known']=len(roster)/5
 out['rookie_experience']=np.mean([np.log1p(seenplayer[x]) for x in line]) if line else 0
 for lag in [1,14]:
  level=[];shift=[];top=[];cov=[];post=[];postcov=[]
  for role in roleorder:
   pl=roster.get(role);counts=Counter(ch for ch,rr,gd,home in hist[pl] if rr==role) if pl else Counter()
   total=sum(counts.values());v=[];dv=[];cc=0;best=[]
   for ch,n in counts.items():
    w=rate(pa,ch,role,day,lag);old=rate(prev.get(pa,''),ch,role,day,lag)
    if w is not None:
     v.append(n*w/max(total,1));cc+=n;best.append(w)
     if old is not None:dv.append(n*(w-old)/max(total,1))
   level.append(sum(v));shift.append(sum(dv));cov.append(cc/max(total,1));top.append(np.mean(sorted(best,reverse=True)[:3]) if best else 0)
   ch=actual.get(role) if actual else None;w=rate(pa,ch,role,day,lag) if ch else None;post.append(w if w is not None else 0);postcov.append(w is not None)
  for name,v in [('patch_level',level),('patch_shift',shift),('patch_options',top),('patch_cover',cov),('draft_wr',post),('draft_cover',postcov)]:out[name+str(lag)]=np.mean(v)
 st=styles.get(team,np.zeros(3));out.update(dict(zip(['tempo','objectives','lane'],st)))
 return out

  v=[];c=[]
  for role in roleorder:
   ch=actual.get(role) if actual else None;w=rate(pa,ch,role,di,lag) if ch else None;v.append(w if w is not None else 0);c.append(w is not None)

 return out

  z=np.array([rr.kills/(rr.gamelength/60), (rr.towers+rr.dragons+rr.barons)/(rr.gamelength/60),rr.gd15/1000 if np.isfinite(rr.gd15) else 0]);styles[rr.teamid]=.9*styles.get(rr.teamid,np.zeros(3))+.1*z
