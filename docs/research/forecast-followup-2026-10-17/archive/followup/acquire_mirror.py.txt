from pathlib import Path
import requests,concurrent.futures,hashlib,json,time
ROOT=Path(__file__).resolve().parent.parent/'prometheus-audit/data/raw'
ROOT.mkdir(parents=True,exist_ok=True)
def fetch(year):
 name=f'{year}_LoL_esports_match_data_from_OraclesElixir.csv'
 if year<2025:url='https://media.githubusercontent.com/media/LouisPoutrain/Lol-ML/69a2856df3a6e566af49473f801b9ad7af8c85ad/raw_data/'+name
 else:url='https://raw.githubusercontent.com/chrisvu1007/league-of-legends-match-predictor/29838abc074d60a83c7a9638a456a2d366b5546d/data/'+name
 p=ROOT/name
 try:
  if not p.exists():
   r=requests.get(url,timeout=(20,45),stream=True);r.raise_for_status()
   it=r.iter_content(1048576);first=next(it)
   if b'gameid' not in first[:1000]:raise ValueError('not raw match CSV')
   tmp=p.with_suffix('.tmp')
   with tmp.open('wb') as f:
    f.write(first)
    for b in it:f.write(b)
   tmp.rename(p)
  h=hashlib.sha256(p.read_bytes()).hexdigest()
  o={'year':year,'url':url,'sha256':h,'bytes':p.stat().st_size}
  print(year,'saved',o['bytes'],flush=True);return o
 except Exception as e:
  print(year,'ERROR',type(e).__name__,str(e),flush=True);return {'year':year,'url':url,'error':str(e)}
out=list(concurrent.futures.ThreadPoolExecutor(max_workers=3).map(fetch,range(2014,2026)))
Path(__file__).with_name('mirror-manifest.json').write_text(json.dumps(out,indent=2))
