rows=pd.read_sql('''SELECT p.gameid,p.teamid,p.playerid,p.position,p.champion,m.date,m.league,m.side,m.result,m.teamname,e.pre_match_elo AS elo,oe.pre_match_elo AS opp_elo FROM match_players p JOIN matches m USING(gameid,teamid) JOIN game_length_elo e USING(gameid,teamid) JOIN game_length_elo oe ON oe.gameid=p.gameid AND oe.teamid!=p.teamid''',c);rows['date']=pd.to_datetime(rows.date);rows=rows[rows.champion.notna()&(rows.champion!='')].sort_values(['date','gameid','teamid','position']).reset_index(drop=True);rows['gd15']=np.nan;rows['p']=1/(1+10**(-(rows.elo-rows.opp_elo)/400));rows['day']=rows.date.dt.normalize();print('rows',len(rows),flush=True)

def prior_day(frame,keys,col):
 a=frame.groupby(keys+['day'],sort=False)[col].sum().sort_index(level='day');p=a.groupby(level=list(range(len(keys)))).cumsum()-a;idx=pd.MultiIndex.from_frame(frame[keys+['day']]);return p.reindex(idx).to_numpy()
def day_features(rows):
 r=rows.copy();r['date']=r.day;r['one']=1.;r['wae']=r.result-r.p;n=prior_day(r,['playerid','champion'],'one');r['exp']=np.log1p(n);r['champ_wae']=prior_day(r,['playerid','champion'],'wae')/(n+10);r['champ_gd15']=0.;r['meta']=d._meta(r)
 op=r[['gameid','teamid','position','champion']].rename(columns={'teamid':'otherteam','champion':'oppchamp'});lanes=r.reset_index().merge(op,on=['gameid','position']);lanes=lanes[lanes.teamid!=lanes.otherteam].set_index('index').sort_index();keys=['position','champion','oppchamp'];nn=prior_day(lanes,keys,'one');r['matchup']=0.;r.loc[lanes.index,'matchup']=prior_day(lanes,keys,'wae')/(nn+20)
 r['token']=r.position+':'+r.champion;t=r.groupby(['gameid','teamid'],sort=False).agg(day=('day','first'),wae=('wae','first'),tokens=('token',list)).reset_index();t=t[t.tokens.map(len)==5];t['pair']=t.tokens.map(lambda x:[a+'|'+b for i,a in enumerate(sorted(x)) for b in sorted(x)[i+1:]]);p=t.explode('pair');p['one']=1.;n=prior_day(p,['pair'],'one');p['value']=prior_day(p,['pair'],'wae')/(n+20);v=p.groupby(['gameid','teamid']).value.sum().rename('synergy');r=r.merge(v,on=['gameid','teamid'],how='left');return r
